cmake_minimum_required(VERSION 3.21)
project(Aoe2ReplayAnalysis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
qt_add_executable(replay-analysis WIN32
    src/main.cpp src/backend.cpp src/eventmodel.cpp src/timeline.cpp src/window.cpp)
target_link_libraries(replay-analysis PRIVATE Qt6::Widgets)
target_compile_definitions(replay-analysis PRIVATE REPLAY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(replay-analysis PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(replay-analysis PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME native-smoke COMMAND replay-analysis --smoke-test)
    set_tests_properties(native-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;PATH=$<TARGET_FILE_DIR:Qt6::Core>")
endif()
install(TARGETS replay-analysis RUNTIME DESTINATION .)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
include(InstallRequiredSystemLibraries)
install(DIRECTORY backend profiles DESTINATION . PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(DIRECTORY licenses DESTINATION .)
install(FILES README.md CONTRACT.md LICENSE THIRD_PARTY.md VALIDATION.md launch.ps1 DESTINATION .)
if(WIN32)
    # Qt 6.8's generated deploy script fails to quote executable paths with spaces.
    find_program(REPLAY_WINDEPLOYQT NAMES windeployqt HINTS "${Qt6_DIR}/../../../bin" REQUIRED)
    install(CODE "execute_process(COMMAND \"${REPLAY_WINDEPLOYQT}\" --release --no-translations --no-opengl-sw --no-system-d3d-compiler \"\${CMAKE_INSTALL_PREFIX}/replay-analysis.exe\" COMMAND_ERROR_IS_FATAL ANY)")
    install(FILES "${Qt6_DIR}/../../../plugins/platforms/qoffscreen.dll" DESTINATION platforms)
else()
    qt_generate_deploy_app_script(TARGET replay-analysis OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()
