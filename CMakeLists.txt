cmake_minimum_required(VERSION 3.21)
project(Aoe2ReplayAnalysis VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
qt_add_executable(replay-analysis WIN32
    src/main.cpp src/backend.cpp src/pythonruntime.cpp src/eventmodel.cpp src/timeline.cpp src/window.cpp)
target_link_libraries(replay-analysis PRIVATE Qt6::Widgets)
if(MSVC)
    target_compile_options(replay-analysis PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(replay-analysis PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
    qt_add_executable(native-regressions tests/test_native.cpp
        src/backend.cpp src/pythonruntime.cpp src/eventmodel.cpp src/timeline.cpp src/window.cpp)
    target_include_directories(native-regressions PRIVATE src)
    set_target_properties(native-regressions PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(native-regressions PRIVATE Qt6::Widgets Qt6::Test)
    add_test(NAME native-regressions COMMAND native-regressions)
    set_tests_properties(native-regressions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;PATH=$<TARGET_FILE_DIR:Qt6::Core>;AOE2_TEST_PYTHON=${Python3_EXECUTABLE};AOE2_TEST_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME native-smoke COMMAND replay-analysis --smoke-test)
    set_tests_properties(native-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;PATH=$<TARGET_FILE_DIR:Qt6::Core>;AOE2_BACKEND_ROOT=${CMAKE_CURRENT_SOURCE_DIR}")
endif()
install(TARGETS replay-analysis RUNTIME DESTINATION .)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
include(InstallRequiredSystemLibraries)
install(DIRECTORY backend profiles DESTINATION . PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(DIRECTORY licenses DESTINATION .)
install(FILES README.md CONTRACT.md LICENSE THIRD_PARTY.md VALIDATION.md launch.ps1 DESTINATION .)
if(WIN32)
    # Qt 6.8's generated deploy script fails to quote executable paths with spaces.
    find_program(REPLAY_WINDEPLOYQT NAMES windeployqt HINTS "${Qt6_DIR}/../../../bin" REQUIRED)
    install(CODE "execute_process(COMMAND \"${REPLAY_WINDEPLOYQT}\" --release --no-translations --no-opengl-sw --no-system-d3d-compiler \"\${CMAKE_INSTALL_PREFIX}/replay-analysis.exe\" COMMAND_ERROR_IS_FATAL ANY)")
    install(FILES "${Qt6_DIR}/../../../plugins/platforms/qoffscreen.dll" DESTINATION platforms)
else()
    qt_generate_deploy_app_script(TARGET replay-analysis OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()
